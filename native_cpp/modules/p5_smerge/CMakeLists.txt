cmake_minimum_required(VERSION 3.20)
project(CTEAS_P5_SMERGE_Scoped LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cteas_p5_tests ../../tests/p5_smerge_tests.cpp)
target_include_directories(cteas_p5_tests PRIVATE ../../include)
if(MSVC)
  target_compile_options(cteas_p5_tests PRIVATE /W4 /WX)
else()
  target_compile_options(cteas_p5_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME p5_generic_utc_input_contracts COMMAND cteas_p5_tests)
